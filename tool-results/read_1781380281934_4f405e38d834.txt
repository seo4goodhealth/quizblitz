    91→// ==================== LANGUAGE SELECTOR COMPONENT ====================
    92→function LanguageSelector({ compact = false, onLocaleChange }: { compact?: boolean; onLocaleChange?: (locale: string) => void }) {
    93→  const { locale, setLocale, t } = useI18n()
    94→
    95→  const handleChange = (code: string) => {
    96→    if (onLocaleChange) {
    97→      onLocaleChange(code)
    98→    } else {
    99→      setLocale(code as any)
   100→    }
   101→  }
   102→
   103→  if (compact) {
   104→    return (
   105→      <div className="flex items-center gap-1">
   106→        {LANGUAGES.map((lang) => (
   107→          <button
   108→            key={lang.code}
   109→            onClick={() => handleChange(lang.code)}
   110→            className={`px-1.5 py-0.5 rounded text-xs transition-all ${
   111→              locale === lang.code
   112→                ? 'bg-purple-600/30 text-purple-300 ring-1 ring-purple-500/50'
   113→                : 'text-gray-500 hover:text-gray-300 hover:bg-white/5'
   114→            }`}
   115→            title={lang.name}
   116→          >
   117→            {lang.flag}
   118→          </button>
   119→        ))}
   120→      </div>
   121→    )
   122→  }
   123→  return (
   124→    <div className="flex flex-wrap items-center justify-center gap-1.5">
   125→      {LANGUAGES.map((lang) => (
   126→        <button
   127→          key={lang.code}
   128→          onClick={() => handleChange(lang.code)}
   129→          className={`flex items-center gap-1 px-2.5 py-1.5 rounded-lg text-xs font-medium transition-all ${
   130→            locale === lang.code
   131→              ? 'bg-purple-600/30 text-white ring-1 ring-purple-500/50'
   132→              : 'bg-white/5 text-gray-400 hover:text-white hover:bg-white/10'
   133→          }`}
   134→        >
   135→          <span>{lang.flag}</span>
   136→          <span>{lang.name}</span>
   137→        </button>
   138→      ))}
   139→    </div>
   140→  )
   141→}
